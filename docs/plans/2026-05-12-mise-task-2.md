# Mise Task 2: Testcontainers Orchestrator Implementation Plan

**Goal:** Build the Testcontainers orchestrator — a Python script that spins up an ephemeral Docker container mirroring the GitHub Actions Linux runner, executes the full pipeline inside it, and tears it down cleanly — wired to a `mise run qa` entry point.

**Architecture:** The orchestrator (`tests/qa_pipeline_orchestrator.py`) uses the `testcontainers` Python library to programmatically manage a `python:3.12-slim` Docker container. It bind-mounts the project directory into `/workspace`, injects the exact environment variables defined in `.github/workflows/openwrt-docs4ai-00-pipeline.yml` (pointing to container-scoped temp directories), then sequentially executes each pipeline script (01 through 08/09) inside the container via `exec_run()`. It halts on the first non-zero exit code, dumps stdout/stderr to the console, and exits with code 1. The container is guaranteed to be torn down on completion or failure via testcontainers' lifecycle management. A `mise.toml` at the repo root defines `qa` as the task alias.

**Tech Stack:** Python 3.12, testcontainers-python, Docker, mise task runner, WSL2 (Windows Docker backend)

def build_container_command() -> list[str]:
    """Build the shell command that the container will execute.
    
    Creates the directory structure, installs dependencies from requirements.txt,
    then executes each pipeline script in order, halting on first failure.
    """
    mkdir_cmd = (
        "mkdir -p /workspace/tmp/pipeline-ci/downloads"
        " /workspace/tmp/pipeline-ci/processed"
        " /workspace/tmp/pipeline-ci/staged"
    )
    install_cmd = "pip install -r /workspace/.github/scripts/requirements.txt -q"
    
    script_commands: list[str] = []
    for script in PIPELINE_SCRIPTS:
        script_path = f"/workspace/{SCRIPTS_DIR_RELATIVE}/{script}"
        script_commands.append(
            f"echo '=== Running {script} ===' && "
            f"python {script_path} || (echo 'FAILED: {script}' && exit 1)"
        )
    
    pipeline_body = " && ".join([mkdir_cmd, install_cmd] + script_commands)
    return ["/bin/bash", "-c", pipeline_body]


def run_pipeline_in_container() -> int:
    """Spin up container, execute pipeline, return exit code.
    
    Returns:
        0 on full success, 1 on any pipeline script failure.
    """
    from testcontainers.core.container import DockerContainer
    
    print("=== QA Pipeline Orchestrator ===")
    print(f"Project root: {PROJECT_ROOT}")
    print(f"Scripts directory: {PROJECT_ROOT / SCRIPTS_DIR_RELATIVE}")
    print()
    
    command = build_container_command()
    
    with DockerContainer("python:3.12-slim") as container:
        container.with_volume_mapping(
            str(PROJECT_ROOT), "/workspace"
        )
        container.with_env("PIPELINE_RUN_DIR", "/workspace/tmp/pipeline-ci")
        container.with_env("WORKDIR", "/workspace/tmp/pipeline-ci/downloads")
        container.with_env("PROCESSED_DIR", "/workspace/tmp/pipeline-ci/processed")
        container.with_env("STAGED_DIR", "/workspace/tmp/pipeline-ci/staged")
        container.with_env("OUTDIR", "/workspace/tmp/pipeline-ci/staged")
        container.with_env("SKIP_AI", "true")
        container.with_env("VALIDATE_MODE", "soft")
        container.with_env("SKIP_WIKI", "true")
        container.with_env("HOME", "/root")
        
        result = container.exec(command)
        exit_code, output = result
        print(output.decode("utf-8", errors="replace"))
        
        if exit_code != 0:
            print(f"\nPipeline FAILED with exit code {exit_code}", file=sys.stderr)
            return 1
        
        print("\n=== Pipeline PASSED ===")
        return 0
```

def _check_docker_available() -> bool:
    """Return True if Docker is running and accessible."""
    import subprocess
    try:
        result = subprocess.run(
            ["docker", "info"],
            capture_output=True,
            text=True,
            timeout=10,
        )
        return result.returncode == 0
    except (FileNotFoundError, subprocess.TimeoutExpired):
        return False


def main() -> int:
    """Entry point for mise run qa."""
    if not _check_docker_available():
        print(
            "Docker is not available. Ensure Docker Desktop is running "
            "or WSL integration is enabled.",
            file=sys.stderr,
        )
        return 2  # distinct exit code for environment issue
    
    return run_pipeline_in_container()


if __name__ == "__main__":
    sys.exit(main())
```
