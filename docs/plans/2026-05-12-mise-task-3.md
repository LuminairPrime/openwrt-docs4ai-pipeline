### Task 1.2: Add container lifecycle (spin up, tear down)

**Step 1: Add testcontainers dependency and container configuration**

```python
from testcontainers.core.container import DockerContainer

CONTAINER_IMAGE = "python:3.12-slim"
CONTAINER_WORKSPACE = "/workspace"

CI_ENV = {
    "PIPELINE_RUN_DIR": f"{CONTAINER_WORKSPACE}/tmp/pipeline-ci",
    "WORKDIR": f"{CONTAINER_WORKSPACE}/tmp/pipeline-ci/downloads",
    "PROCESSED_DIR": f"{CONTAINER_WORKSPACE}/tmp/pipeline-ci/processed",
    "STAGED_DIR": f"{CONTAINER_WORKSPACE}/tmp/pipeline-ci/staged",
    "OUTDIR": f"{CONTAINER_WORKSPACE}/tmp/pipeline-ci/staged",
}

def create_container() -> DockerContainer:
    container = (
        DockerContainer(CONTAINER_IMAGE)
        .with_bind_mount(str(PROJECT_ROOT), CONTAINER_WORKSPACE)
        .with_env("CI", "true")
        .with_command("tail -f /dev/null")
    )
    for key, value in CI_ENV.items():
        container = container.with_env(key, value)
    return container

def _exec(container, cmd: str) -> tuple[int, str]:
    result = container.exec(cmd)
    return result.exit_code, (result.output or b"").decode("utf-8", errors="replace")
```

**Step 2: Wire container lifecycle into main()**

```python
def main() -> int:

    try:

        _exec(container, "mkdir -p /workspace/tmp/pipeline-ci/processed")
```

### Task 2.2: Implement sequential execution with exit-code validation

**Files:**
- Modify: tests/qa_pipeline_orchestrator.py

**Step 1: Add the pipeline runner function**

```python
def run_pipeline(container, stages: list[PipelineStage]) -> int:
    """Execute stages sequentially, halting on first failure.
    Returns 0 on success, 1 on failure."""
    for stage in stages:
        script = _script_path(stage)
        print(f"\n{'='*60}")
        print(f"Stage: {stage.label} ({stage.script})")
        print(f"{'='*60}")
        
        cmd = f"cd {CONTAINER_WORKSPACE} && python {script}"
        exit_code, output = _exec(container, cmd)
        
        if exit_code != 0:
            if stage.optional:
                print(f"[WARN] Optional stage {stage.script} failed (exit {exit_code}), continuing...")
            else:
                print(f"[FAIL] Stage {stage.script} FAILED with exit code {exit_code}")
                print("--- STDOUT/STDERR ---")
                print(output[-4000:] if len(output) > 4000 else output)
                print("--- END OUTPUT ---")
                return 1
        else:
            print(f"[PASS] Stage {stage.script} completed successfully")
    return 0
```

### Task 3.2: Add CLI argument support to orchestrator

**Files:**
- Modify: tests/qa_pipeline_orchestrator.py

**Step 1: Add argument parsing**

```python
import argparse

def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="QA Pipeline Orchestrator - CI-in-a-box via Testcontainers"
    )
    parser.add_argument(
        "--smoke",
        action="store_true",
        help="Run smoke-only: skip extractors (01, 02a-02i) and AI (04)"
    )
    parser.add_argument(
        "--stage",
        type=str,
        help="Run a single stage by script name (e.g. '08-validate-output')"
    )
    return parser.parse_args()

def active_stages(args: argparse.Namespace) -> list[PipelineStage]:
    if args.stage:
        return [s for s in PIPELINE_STAGES if s.script == args.stage]
    if args.smoke:
        return [s for s in PIPELINE_STAGES if not s.script.startswith("0")]
    return PIPELINE_STAGES
```

**Step 2: Wire argument parsing into main()**

```python
def main() -> int:
    args = parse_args()
    stages = active_stages(args)
    print(f"QA Pipeline Orchestrator starting ({len(stages)} stages)...")
    container = create_container()
    try:
        print(f"Starting container ({CONTAINER_IMAGE})...")
        container.start()
        _exec(container, "mkdir -p /workspace/tmp/pipeline-ci/downloads")
        _exec(container, "mkdir -p /workspace/tmp/pipeline-ci/processed")
        _exec(container, "mkdir -p /workspace/tmp/pipeline-ci/staged")
        print("Container ready.")
        return run_pipeline(container, stages)
    finally:
        print("Tearing down container...")
        container.stop()
        print("Container destroyed.")
```
